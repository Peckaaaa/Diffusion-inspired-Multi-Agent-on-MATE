#   conda activate dima
#   pip install -r requirements.txt          # run from the repository root

# torch is pinned to a CPU build, which is what this project was verified on.
# For CUDA instead, install torch/torchvision from the matching PyTorch index
# BEFORE this file, e.g.
#   pip install torch==2.13.0 torchvision==0.28.0 --index-url https://download.pytorch.org/whl/cu121

#   pip install torch==2.13.0 torchvision==0.28.0 --index-url https://download.pytorch.org/whl/cpu

# NOTE: `DIMA/environment.yml` is the *upstream DIMA* conda file.  It pins
